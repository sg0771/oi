/**
 * Problem: T2 游戏 (game)
 * Algorithm: 博弈论 + 贪心排序
 * Time Complexity: O(n log n)
 * Space Complexity: O(n)
 */

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long LL;

int main() {
    // 提升 I/O 吞吐性能
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) return 0;

    vector<LL> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    // 将卡片权值按从大到小降序排序
    // 两人都追求自身最优，每步必然优先选择当前剩余的最大数值
    sort(a.begin(), a.end(), greater<LL>());

    LL sumA = 0; // 小 A 的卡片总和
    LL sumB = 0; // 小 B 的卡片总和

    for (int i = 0; i < n; ++i) {
        if (i % 2 == 0) {
            // 轮到小 A（第 0, 2, 4... 张）
            sumA += a[i];
        } else {
            // 轮到小 B（第 1, 3, 5... 张）
            sumB += a[i];
        }
    }

    // 输出小 A 的最终净得分 A - B
    cout << sumA - sumB << "\n";

    return 0;
}
