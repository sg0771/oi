# P8820 [CSP-S 2022] 数据传输 题解

## 题目大意

一棵 $n$ 个节点的树，节点 $i$ 有权值 $v_i$。数据可以在距离 $\le k$ 的节点之间直接传输（$k \in \{1,2,3\}$），传输代价为经过节点的权值之和。$q$ 次询问从 $s$ 到 $t$ 的最小传输代价。

---

## 解题思路

### 核心观察

当 $k=1$ 时，只能沿树边传输，答案就是路径上所有节点权值之和。

当 $k=2$ 时，每次传输可以跨越 $2$ 条边，意味着可以跳过路径上的某些节点。此时路径上的最优方案可能利用路径外但与路径相邻的节点（代价更小的节点）来"绕路"。

当 $k=3$ 时，可以跨越 $3$ 条边，情况更复杂。

### 算法：倍增 DP

定义状态（以 $k=2$ 为例，$K=2$）：
- 状态 $0$：当前在路径节点上
- 状态 $1$：当前在路径节点的某个子节点上（不在路径上，但与路径相邻）

对于每个节点 $u$，预处理：
- $g[u][0] = v_u$
- $g[u][1] = \min(v_u, \min_{c \in \text{children}(u)} v_c)$（$k \ge 2$）
- $g[u][2] = \min(g[u][1], \min_{c \in \text{children}(u)} g[c][1])$（$k = 3$）

定义 $f[u][i][a][b]$ 表示从节点 $u$（处于状态 $a$）到其 $2^i$ 级祖先（处于状态 $b$）的最小代价。

**初始转移**（$i=0$，父节点 $p$）：
$$f[u][0][a][b] = g[u][a] + g[p][b]$$

**矩阵合并**（从 $u$ 到 $2^i$ 级祖先 $anc$）：
$$f[u][i][a][b] = \min_{c} \big( f[u][i-1][a][c] + f[anc][i-1][c][b] - g[anc][c] \big)$$

其中减去 $g[anc][c]$ 是因为中间节点 $anc$ 在两端都被计入，需要去重。

**查询**时，将 $s$ 和 $t$ 分别跳到 LCA，沿途维护两个"当前状态向量"（到达当前位置在各状态下的最小代价），最后在 LCA 处合并。

## C++ 源代码

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll INF = 1e18;
const int MAXN = 200005;
const int LOG = 18;

int n, q, K;
ll v[MAXN];
vector<int> adj[MAXN];

// 倍增祖先
int up[MAXN][LOG];
int depth[MAXN];

// g[u][s]: 节点 u 在状态 s 下的代价
ll g[MAXN][3];

// f[u][i][a][b]: 从 u 到 2^i 级祖先，状态 a -> b 的最小代价
ll f[MAXN][LOG][3][3];

// ==================== 预处理 ====================

void dfs(int u, int p) {
    up[u][0] = p;
    depth[u] = depth[p] + 1;
    for (int i = 1; i < LOG; i++) {
        up[u][i] = up[up[u][i-1]][i-1];
    }

    // 计算 g[u][s]
    g[u][0] = v[u];
    if (K >= 2) {
        g[u][1] = v[u];
        for (int w : adj[u]) {
            g[u][1] = min(g[u][1], v[w]);
        }
    }
    if (K >= 3) {
        g[u][2] = g[u][1];
        for (int w : adj[u]) {
            if (w == p) continue;
            for (int x : adj[w]) {
                g[u][2] = min(g[u][2], v[x]);
            }
        }
    }

    // 初始化 f[u][0][a][b]
    if (p != 0) {
        for (int a = 0; a < K; a++) {
            for (int b = 0; b < K; b++) {
                f[u][0][a][b] = g[u][a] + g[p][b];
            }
        }
    }

    for (int w : adj[u]) {
        if (w != p) dfs(w, u);
    }
}

// 合并两个矩阵
void merge_mat(ll C[3][3], const ll A[3][3], const ll B[3][3], int mid_node) {
    for (int a = 0; a < K; a++) {
        for (int b = 0; b < K; b++) {
            C[a][b] = INF;
            for (int c = 0; c < K; c++) {
                C[a][b] = min(C[a][b], A[a][c] + B[c][b] - g[mid_node][c]);
            }
        }
    }
}

void build() {
    dfs(1, 0);
    for (int i = 1; i < LOG; i++) {
        for (int u = 1; u <= n; u++) {
            int mid = up[u][i-1];
            if (mid == 0) continue;
            merge_mat(f[u][i], f[u][i-1], f[mid][i-1], mid);
        }
    }
}

// ==================== 查询 ====================

// 将节点 u 跳到深度 d，同时维护当前状态向量 dp（长度为 K）
// dp[s] 表示从起点到达 u 的当前状态 s 的最小代价
void lift(int &u, int target_depth, ll dp[3]) {
    int diff = depth[u] - target_depth;
    for (int i = LOG - 1; i >= 0; i--) {
        if (diff & (1 << i)) {
            ll ndp[3];
            fill(ndp, ndp + K, INF);
            for (int a = 0; a < K; a++) {
                for (int b = 0; b < K; b++) {
                    ndp[b] = min(ndp[b], dp[a] + f[u][i][a][b] - g[u][a]);
                }
            }
            for (int s = 0; s < K; s++) dp[s] = ndp[s];
            u = up[u][i];
            diff -= (1 << i);
        }
    }
}

ll query(int s, int t) {
    if (depth[s] < depth[t]) swap(s, t);

    // dp_s 和 dp_t 分别表示从 s 和 t 出发到达当前位置各状态的最小代价
    ll dp_s[3], dp_t[3];
    fill(dp_s, dp_s + K, INF);
    fill(dp_t, dp_t + K, INF);
    dp_s[0] = v[s];
    dp_t[0] = v[t];

    // 将 s 跳到与 t 同深度
    lift(s, depth[t], dp_s);

    if (s == t) {
        // s 和 t 相遇：dp_s 和 dp_t 都到达 s(=t)，合并
        ll ans = INF;
        for (int a = 0; a < K; a++) {
            ans = min(ans, dp_s[a] + dp_t[a] - g[s][a]);
        }
        return ans;
    }

    // 同时跳 s 和 t
    for (int i = LOG - 1; i >= 0; i--) {
        if (up[s][i] != up[t][i]) {
            // 跳 s
            ll ndp_s[3];
            fill(ndp_s, ndp_s + K, INF);
            for (int a = 0; a < K; a++) {
                for (int b = 0; b < K; b++) {
                    ndp_s[b] = min(ndp_s[b], dp_s[a] + f[s][i][a][b] - g[s][a]);
                }
            }
            for (int a = 0; a < K; a++) dp_s[a] = ndp_s[a];
            // 跳 t
            ll ndp_t[3];
            fill(ndp_t, ndp_t + K, INF);
            for (int a = 0; a < K; a++) {
                for (int b = 0; b < K; b++) {
                    ndp_t[b] = min(ndp_t[b], dp_t[a] + f[t][i][a][b] - g[t][a]);
                }
            }
            for (int a = 0; a < K; a++) dp_t[a] = ndp_t[a];

            s = up[s][i];
            t = up[t][i];
        }
    }

    // s 和 t 的父亲是 LCA
    int lca = up[s][0];

    // 将 s 和 t 跳到 lca
    ll ndp_s[3], ndp_t[3];
    fill(ndp_s, ndp_s + K, INF);
    fill(ndp_t, ndp_t + K, INF);
    for (int a = 0; a < K; a++) {
        for (int b = 0; b < K; b++) {
            ndp_s[b] = min(ndp_s[b], dp_s[a] + f[s][0][a][b] - g[s][a]);
            ndp_t[b] = min(ndp_t[b], dp_t[a] + f[t][0][a][b] - g[t][a]);
        }
    }

    // 在 lca 处合并
    ll ans = INF;
    for (int a = 0; a < K; a++) {
        ans = min(ans, ndp_s[a] + ndp_t[a] - g[lca][a]);
    }
    return ans;
}

// ==================== 主函数 ====================

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> q >> K;
    for (int i = 1; i <= n; i++) cin >> v[i];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    build();

    while (q--) {
        int s, t;
        cin >> s >> t;
        cout << query(s, t) << "\n";
    }

    return 0;
}
```

## 复杂度分析

- **时间复杂度**：$O((n + q) \log n \cdot K^3)$，其中 $K \in \{1,2,3\}$ 是状态数。预处理 $O(n \log n \cdot K^2)$，每次查询 $O(\log n \cdot K^3)$。
- **空间复杂度**：$O(n \log n \cdot K^2)$，存储倍增 DP 矩阵。

---

## 关键点

1. **状态定义**：状态 $0$ 表示在路径节点上，状态 $1$ 表示在路径节点的子节点上（$k \ge 2$），状态 $2$ 表示在路径节点的孙子节点上（$k = 3$）。
2. **矩阵合并**：合并两个倍增段时，中间节点会被重复计算，需要减去 $g[\text{mid}][c]$。
3. **查询合并**：$s$ 和 $t$ 分别跳到 LCA 后，在 LCA 处合并两个状态向量，注意减去重复计算的 LCA 代价。
4. **$k=1$ 特判**：当 $k=1$ 时状态数 $K=1$，答案退化为路径上节点权值之和。
