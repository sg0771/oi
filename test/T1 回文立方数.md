/**
 * Problem: T1 回文立方数 (cube) - 打表二分最优解
 * Time Complexity: O(log 14) ≈ O(1)
 * Space Complexity: O(1)
 */

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef unsigned long long ULL;

// 预先打出 10^18 内所有回文立方数（共 14 个）
const vector<ULL> PAL_CUBES = {
    1ULL,
    8ULL,
    343ULL,
    1331ULL,
    1030301ULL,
    1367631ULL,
    1003003001ULL,
    10662526601ULL,
    1000300030001ULL,
    1030607060301ULL,
    1334996994331ULL,
    1000030000300001ULL,
    1033394994933301ULL,
    1331399339931331ULL
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    ULL n;
    if (!(cin >> n)) return 0;

    // 二分查找第一个 > n 的位置，其前一个元素就是 <= n 的最大值
    auto it = upper_bound(PAL_CUBES.begin(), PAL_CUBES.end(), n);
    
    // 题目保证 1 <= n <= 10^18，因此必定存在解
    if (it != PAL_CUBES.begin()) {
        --it;
        cout << *it << "\n";
    }

    return 0;
}
